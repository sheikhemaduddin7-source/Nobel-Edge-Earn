                    className="px-3.5 py-2 rounded-lg bg-navy-900 hover:bg-navy-800 text-gold-400 border border-gold-500/30 transition flex items-center space-x-2 shadow"
                  >
                    <i data-lucide="lock" className="w-3.5 h-3.5 text-gold-400"></i>
                    <span>Launch Admin App</span>
                  </button>
                  <p className="text-[11px] text-slate-500 mt-2">Restricted separate administrative portal.</p>
                </div>
              </div>

              <div className="pt-6 border-t border-navy-900 text-center text-slate-500">
                © {new Date().getFullYear()} Creative Skills Academy. All rights reserved.
              </div>
            </div>
          </footer>

          {/* ========================================================================= */}
          {/* CERTIFICATE MODAL (WITH EXACT LOCATION & DYNAMIC TODAY'S DATE) */}
          {/* ========================================================================= */}
          {certModal.open && certModal.data && (
            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-navy-950/80 backdrop-blur-sm overflow-y-auto">
              <div className="bg-white rounded-xl shadow-2xl max-w-4xl w-full p-4 text-slate-900 my-8">
                <div className="flex justify-between items-center pb-3 mb-3 border-b border-slate-200">
                  <span className="text-xs font-bold uppercase text-navy-900">Official Certificate of Completion</span>
                  <div className="flex items-center space-x-2">
                    <button 
                      onClick={() => window.print()}
                      className="px-3 py-1.5 bg-navy-900 text-gold-400 rounded text-xs font-bold"
                    >
                      Print Certificate
                    </button>
                    <button 
                      onClick={() => setCertModal({ open: false, data: null })}
                      className="p-1 text-slate-400 hover:text-slate-700"
                    >
                      <i data-lucide="x" className="w-5 h-5"></i>
                    </button>
                  </div>
                </div>

                <div id="printable-certificate" className="bg-[#FAF9F6] p-8 border-4 border-navy-900 text-center font-serif text-navy-950">
                  <div className="border border-gold-500 p-6">
                    <h2 className="text-xs uppercase tracking-widest font-sans font-bold text-navy-900">Creative Skills Academy</h2>
                    <p className="text-[10px] text-gold-600 uppercase font-sans mb-4">Learn • Practice • Excel</p>

                    <h1 className="text-3xl font-extrabold uppercase text-navy-900 border-b border-gold-500/40 pb-2 inline-block mb-3">
                      Certificate of Completion
                    </h1>

                    <p className="font-sans text-xs uppercase tracking-widest text-slate-500 mt-2">
                      This is to certify that
                    </p>

                    {/* Recipient Name and Location */}
                    <div className="my-3">
                      <span className="text-2xl sm:text-3xl font-bold text-gold-600 font-serif border-b border-slate-300 px-6 pb-1 inline-block">
                        {certModal.data.name}
                      </span>
                      {certModal.data.location && (
                        <p className="font-sans text-xs text-slate-600 mt-2 font-medium">
                          <span className="text-slate-400">Hometown / Address:</span> <strong className="text-navy-900">{certModal.data.location}</strong>
                        </p>
                      )}
                    </div>

                    <p className="font-sans text-xs text-slate-700 max-w-lg mx-auto leading-relaxed mt-3">
                      has successfully completed the <strong>{certModal.data.course}</strong> and demonstrated skills in graphic design, visual composition, and digital creativity.
                    </p>

                    <div className="mt-8 pt-4 grid grid-cols-2 font-sans text-xs">
                      <div>
                        <div className="w-28 mx-auto border-b border-navy-900 pb-1 italic font-serif">M. Armstrong</div>
                        <span className="text-[10px] text-slate-500 uppercase mt-1 block">Instructor Signature</span>
                      </div>
                      <div>
                        <div className="w-28 mx-auto border-b border-navy-900 pb-1 italic font-serif">Director Council</div>
                        <span className="text-[10px] text-slate-500 uppercase mt-1 block">Director Signature</span>
                      </div>
                    </div>

                    {/* সার্টিফিকেট এ লোকেশন, আইডি এবং আজকের তারিখ */}
                    <div className="mt-6 pt-3 border-t border-slate-200 flex justify-between items-center text-[10px] font-sans text-slate-500">
                      <span>ID: <strong>{certModal.data.certId}</strong></span>
                      <span>LOCATION: <strong>{certModal.data.location}</strong></span>
                      <span>DATE: <strong>{certModal.data.issueDate || getTodayDateFormatted()}</strong></span>
                    </div>
                  </div>
                </div>
              </div>
            </div>
          )}

          {/* AUTH MODAL */}
          {authModal.open && (
            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-navy-950/80">
              <div className="bg-white dark:bg-navy-900 rounded-xl shadow-xl max-w-sm w-full p-6 text-slate-800 dark:text-slate-100 border border-slate-200 dark:border-navy-800">
                <div className="flex justify-between items-center mb-4">
                  <h3 className="text-base font-bold text-navy-900 dark:text-white">
                    {authModal.mode === 'login' ? 'Student Login' : 'Admission Application'}
                  </h3>
                  <button onClick={() => setAuthModal({ open: false, mode: 'login' })} className="text-slate-400">
                    <i data-lucide="x" className="w-4 h-4"></i>
                  </button>
                </div>

                <form onSubmit={(e) => {
                  e.preventDefault();
                  if (authModal.mode === 'login') {
                    setCurrentUser({ name: "Muhammad Imad Uddin", email: "imad.uddin@example.com" });
                    showToast("Logged in as Muhammad Imad Uddin");
                    setCurrentRoute('student-portal');
                  } else {
                    showToast("Admission submitted successfully");
                  }
                  setAuthModal({ open: false, mode: 'login' });
                }} className="space-y-3">

                  {authModal.mode === 'register' && (
                    <div>
                      <label className="block text-xs font-bold text-slate-500 mb-1">Full Name</label>
                      <input required type="text" placeholder="Your Full Name" className="w-full px-3 py-2 rounded bg-surface-light dark:bg-navy-950 border border-slate-300 dark:border-navy-700 text-xs" />
                    </div>
                  )}

                  <div>
                    <label className="block text-xs font-bold text-slate-500 mb-1">Email</label>
                    <input required type="email" placeholder="name@example.com" className="w-full px-3 py-2 rounded bg-surface-light dark:bg-navy-950 border border-slate-300 dark:border-navy-700 text-xs" />
                  </div>

                  <div>
                    <label className="block text-xs font-bold text-slate-500 mb-1">Password</label>
                    <input required type="password" placeholder="••••••••" className="w-full px-3 py-2 rounded bg-surface-light dark:bg-navy-950 border border-slate-300 dark:border-navy-700 text-xs" />
                  </div>

                  <button type="submit" className="w-full py-2.5 rounded bg-gold-500 hover:bg-gold-400 text-navy-950 font-bold text-xs mt-2">
                    {authModal.mode === 'login' ? 'Sign In' : 'Submit Admission'}
                  </button>
                </form>

                <div className="mt-3 text-center text-xs text-slate-500">
                  {authModal.mode === 'login' ? (
                    <button onClick={() => setAuthModal({ open: true, mode: 'register' })} className="text-gold-500 hover:underline">Apply for Admission</button>
                  ) : (
                    <button onClick={() => setAuthModal({ open: true, mode: 'login' })} className="text-gold-500 hover:underline">Already registered? Log in</button>
                  )}
                </div>
              </div>
            </div>
          )}

        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
